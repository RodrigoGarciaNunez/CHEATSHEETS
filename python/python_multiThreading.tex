\section*{Módulo}
\texttt{threading}

\section*{Thread básico}
\begin{lstlisting}
import threading

def f():
    pass

t = threading.Thread(target=f)
t.start()
t.join()
\end{lstlisting}

\section*{Lock}
\begin{lstlisting}
lock = threading.Lock()
with lock:
    pass
\end{lstlisting}
