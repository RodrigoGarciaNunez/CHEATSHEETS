\section*{¿Qué es NGINX?}

NGINX es:
\begin{itemize}
    \item Web Server
    \item Reverse Proxy
    \item Load Balancer
    \item API Gateway
    \item Proxy para contenedores
\end{itemize}

Muy usado con:
\begin{itemize}
    \item Docker
    \item Kubernetes
    \item NodeJS
    \item FastAPI
    \item Flask
\end{itemize}

\section*{nginx.conf básico}

\begin{lstlisting}
events {}

http {

    server {

        listen 80;

        server_name localhost;

        location / {

            root /var/www/html;

            index index.html;

        }
    }
}
\end{lstlisting}

\section*{Reverse Proxy}

\begin{lstlisting}
location /api/ {

    proxy_pass http://localhost:5000/;

}
\end{lstlisting}

NGINX recibe requests y las reenvía.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Headers importantes}

\begin{lstlisting}
location / {

    proxy_pass http://backend;

    proxy_set_header Host $host;

    proxy_set_header X-Real-IP $remote_addr;

    proxy_set_header X-Forwarded-For
    $proxy_add_x_forwarded_for;

}
\end{lstlisting}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Load Balancer}

\begin{lstlisting}
upstream backend {

    server localhost:5000;

    server localhost:5001;

}

server {

    location / {

        proxy_pass http://backend;

    }
}
\end{lstlisting}

\section*{Proxy para WebSockets}

\begin{lstlisting}
location /socket {

    proxy_pass http://backend;

    proxy_http_version 1.1;

    proxy_set_header Upgrade $http_upgrade;

    proxy_set_header Connection "upgrade";

}
\end{lstlisting}
